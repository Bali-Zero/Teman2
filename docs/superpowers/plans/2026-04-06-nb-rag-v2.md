## Task 3: Build Golden Answers Refresh Script

**Files:**
- Create: `apps/backend-rag/scripts/golden_answers_refresh.py`

- [ ] **Step 1: Create the refresh script**

This script reads `golden_answers_questions.yaml`, queries NLM for each question, and populates the `golden_answers` + `query_clusters` PostgreSQL tables.

```python
#!/usr/bin/env python3
"""
Golden Answers Refresh Script

Reads canonical questions from YAML, queries NLM notebooks for grounded answers,
and populates the golden_answers + query_clusters PostgreSQL tables.

Usage:
    PYTHONPATH=. python scripts/golden_answers_refresh.py [--dry-run] [--domain visa]

Designed to run as weekly cron: Sunday 03:00 WITA
"""

from __future__ import annotations

import argparse
import asyncio
import hashlib
import logging
import os
import sys
import time
from pathlib import Path
from typing import Any

import asyncpg
import yaml

# Add project root to path
sys.path.insert(0, str(Path(__file__).parent.parent))

from apps.evaluator.nlm_deep_research.nlm_bridge import check_nlm_available, nlm_query

logging.basicConfig(level=logging.INFO, format="%(asctime)s %(levelname)s %(message)s")
logger = logging.getLogger(__name__)

QUESTIONS_FILE = Path(__file__).parent / "golden_answers_questions.yaml"
DEFAULT_TIMEOUT = 120
MAX_RETRIES = 2


async def load_questions(domain_filter: str | None = None) -> list[dict[str, Any]]:
    """Load canonical questions from YAML, optionally filtered by domain."""
    with open(QUESTIONS_FILE) as f:
        data = yaml.safe_load(f)

    questions = data.get("questions", [])
    if domain_filter:
        questions = [q for q in questions if q["domain"] == domain_filter]

    logger.info(f"Loaded {len(questions)} questions (filter={domain_filter or 'all'})")
    return questions


def generate_cluster_id(question: str) -> str:
    """Generate deterministic cluster_id from question text."""
    return hashlib.sha256(question.lower().strip().encode("utf-8")).hexdigest()[:16]


async def query_nlm_for_answer(
    question: str, nb_id: str, retries: int = MAX_RETRIES
) -> dict[str, Any] | None:
    """Query NLM notebook and return structured answer."""
    for attempt in range(retries + 1):
        try:
            result = nlm_query(
                notebook_id=nb_id,
                query=question,
                timeout=DEFAULT_TIMEOUT,
            )

            if result.get("status") == "success" and result.get("answer"):
                return {
                    "answer": result["answer"],
                    "sources": ", ".join(result.get("sources_used", [])),
                    "citations": result.get("citations", {}),
                }

            logger.warning(f"NLM returned no answer for: {question[:60]}... (attempt {attempt+1})")

        except Exception as e:
            logger.error(f"NLM query failed: {e} (attempt {attempt+1})")
            if attempt < retries:
                await asyncio.sleep(5)

    return None


async def upsert_golden_answer(
    pool: asyncpg.Pool,
    cluster_id: str,
    question: str,
    answer: str,
    sources: str,
    domain: str,
    aliases: list[str],
) -> None:
    """Upsert golden answer and its query cluster aliases."""
    async with pool.acquire() as conn:
        # Upsert golden answer
        await conn.execute(
            """
            INSERT INTO golden_answers (cluster_id, canonical_question, answer, sources, confidence, usage_count, created_at, last_used_at)
            VALUES ($1, $2, $3, $4, 0.85, 0, NOW(), NOW())
            ON CONFLICT (cluster_id) DO UPDATE SET
                canonical_question = $2,
                answer = $3,
                sources = $4,
                created_at = NOW()
            """,
            cluster_id, question, answer, sources,
        )

        # Insert alias query_clusters (hash of each alias → cluster_id)
        all_queries = [question] + aliases
        for q in all_queries:
            query_hash = hashlib.md5(q.lower().strip().encode("utf-8")).hexdigest()
            await conn.execute(
                """
                INSERT INTO query_clusters (query_hash, cluster_id, query_text)
                VALUES ($1, $2, $3)
                ON CONFLICT (query_hash) DO UPDATE SET cluster_id = $2, query_text = $3
                """,
                query_hash, cluster_id, q,
            )


async def main(dry_run: bool = False, domain_filter: str | None = None) -> None:
    """Main refresh flow."""
    # Validate NLM CLI
    if not check_nlm_available():
        logger.error("NLM CLI not available. Run: npm install -g notebooklm-mcp")
        sys.exit(1)

    questions = await load_questions(domain_filter)
    if not questions:
        logger.warning("No questions to process")
        return

    database_url = os.environ.get("DATABASE_URL")
    if not database_url:
        logger.error("DATABASE_URL not set")
        sys.exit(1)

    pool = await asyncpg.create_pool(database_url, min_size=1, max_size=3)

    stats = {"total": len(questions), "success": 0, "failed": 0, "skipped": 0}
    start = time.time()

    for i, q in enumerate(questions, 1):
        question = q["question"]
        nb_id = q["nb_id"]
        domain = q["domain"]
        aliases = q.get("aliases", [])
        cluster_id = generate_cluster_id(question)

        logger.info(f"[{i}/{stats['total']}] {domain}: {question[:60]}...")

        if dry_run:
            logger.info(f"  [DRY RUN] Would query NB {nb_id} and upsert cluster {cluster_id}")
            stats["skipped"] += 1
            continue

        result = await query_nlm_for_answer(question, nb_id)

        if result:
            await upsert_golden_answer(
                pool, cluster_id, question,
                result["answer"], result["sources"],
                domain, aliases,
            )
            stats["success"] += 1
            logger.info(f"  OK — {len(result['answer'])} chars, sources: {result['sources'][:80]}")
        else:
            stats["failed"] += 1
            logger.warning(f"  FAILED — no answer from NLM")

        # Rate limit: 2s between NLM queries
        await asyncio.sleep(2)

    elapsed = time.time() - start
    await pool.close()

    logger.info(f"\n{'='*60}")
    logger.info(f"Golden Answers Refresh Complete ({elapsed:.1f}s)")
    logger.info(f"  Total: {stats['total']}")
    logger.info(f"  Success: {stats['success']}")
    logger.info(f"  Failed: {stats['failed']}")
    logger.info(f"  Skipped: {stats['skipped']}")
    logger.info(f"{'='*60}")


if __name__ == "__main__":
    parser = argparse.ArgumentParser(description="Refresh golden answers from NLM")
    parser.add_argument("--dry-run", action="store_true", help="Print plan without executing")
    parser.add_argument("--domain", type=str, default=None, help="Filter by domain (visa, tax, company, property)")
    args = parser.parse_args()

    asyncio.run(main(dry_run=args.dry_run, domain_filter=args.domain))
```

```python
#!/usr/bin/env python3
"""
Add alias metadata to visa_oracle Qdrant collection.
Maps old visa names (B211A, KITAS) to new codes (C316, E-series).

Usage:
    PYTHONPATH=. python scripts/add_visa_aliases.py [--dry-run]
"""

from __future__ import annotations

import asyncio
import json
import logging
import os
import sys
from pathlib import Path

from qdrant_client import AsyncQdrantClient
from qdrant_client.models import SetPayloadOperation, PointIdsList

sys.path.insert(0, str(Path(__file__).parent.parent))

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

# Mapping: visa_code → list of aliases (old names, common misspellings, categories)
VISA_ALIAS_MAP: dict[str, list[str]] = {
    # Visit Visas (old B211 series)
    "B1": ["B211A", "B211", "visit visa", "visa kunjungan", "wisata"],
    "C6": ["social visa", "visa sosial", "social budaya", "B211A social"],
    "C7": ["cultural visa", "visa budaya", "seni budaya", "B211A budaya"],
    "C2": ["business visa", "visa bisnis", "B211 business"],
    "C1": ["tourism visa", "visa wisata", "tourist visa", "VOA", "visa on arrival"],
    "C18": ["work trial visa", "visa uji coba kerja"],
    "C22A": ["digital nomad visit", "visa nomad digital"],

    # KITAS/ITAS (old KITAS series)
    "E23": ["employee KITAS", "work permit", "KITAS kerja", "working KITAS"],
    "E25B": ["director KITAS", "KITAS direktur", "direksi KITAS"],
    "E28A": ["investor KITAS", "KITAS investor", "investor visa"],
    "E28B": ["investor company setup", "KITAS investor pendirian"],
    "E33": ["second home visa", "visa rumah kedua", "E33 second home"],
    "E33E": ["retirement visa", "visa pensiun", "retirement KITAS", "E33E retirement"],
    "E33F": ["retirement visa 55+", "visa lansia"],
    "E33G": ["digital nomad KITAS", "remote worker visa", "E33G digital nomad"],
    "E32A": ["spouse KITAS", "KITAS pasangan", "dependent visa"],

    # KITAP
    "KITAP": ["permanent stay", "izin tinggal tetap", "permanent resident"],

    # Golden Visa
    "GOLDEN_VISA": ["golden visa", "golden visa Indonesia", "KITAP golden", "golden visa 5 year", "golden visa 10 year"],

    # Multiple Entry
    "D12": ["multiple entry visa", "visa kunjungan beberapa kali", "VKMK"],
}


async def enrich_visa_aliases(dry_run: bool = False) -> None:
    """Add aliases to visa_oracle collection payloads."""
    qd = AsyncQdrantClient(
        url=os.environ["QDRANT_URL"],
        api_key=os.environ["QDRANT_API_KEY"],
    )

    collection = "visa_oracle"
    info = await qd.get_collection(collection)
    logger.info(f"visa_oracle: {info.points_count} points")

    # Scroll through all points
    updated = 0
    offset = None

    while True:
        points, offset = await qd.scroll(
            collection_name=collection,
            limit=100,
            offset=offset,
            with_payload=True,
        )

        if not points:
            break

        for point in points:
            payload = point.payload or {}
            metadata = payload.get("metadata", {})
            visa_code = metadata.get("code", "")

            if visa_code in VISA_ALIAS_MAP:
                aliases = VISA_ALIAS_MAP[visa_code]
                alias_text = ", ".join(aliases)

                if dry_run:
                    logger.info(f"  [DRY RUN] {visa_code} → aliases: {alias_text}")
                else:
                    await qd.set_payload(
                        collection_name=collection,
                        payload={"aliases": alias_text},
                        points=[point.id],
                    )
                updated += 1

        if offset is None:
            break

    logger.info(f"Updated {updated} points with aliases (dry_run={dry_run})")


if __name__ == "__main__":
    dry_run = "--dry-run" in sys.argv
    asyncio.run(enrich_visa_aliases(dry_run=dry_run))
```

```python
#!/usr/bin/env python3
"""
Enrich immigration_circulars collection by extracting relevant chunks
from legal_unified_hybrid_hybrid.

Targets: Golden Visa, Digital Nomad, Second Home, KITAS, KITAP circulars.

Usage:
    PYTHONPATH=. python scripts/enrich_immigration_circulars.py [--dry-run]
"""

from __future__ import annotations

import asyncio
import logging
import os
import sys
import uuid
from pathlib import Path

from qdrant_client import AsyncQdrantClient
from qdrant_client.models import PointStruct

sys.path.insert(0, str(Path(__file__).parent.parent))

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

# Keywords to match immigration-relevant chunks from legal_unified
IMMIGRATION_KEYWORDS = [
    "golden visa", "second home", "digital nomad", "remote worker",
    "visa tinggal terbatas", "visa kunjungan", "izin tinggal",
    "KITAS", "KITAP", "VITAS", "E-VOA", "e-visa",
    "imigrasi", "keimigrasian", "izin masuk kembali",
    "orang asing", "tenaga kerja asing", "TKA", "RPTKA", "IMTA",
    "Permenkumham", "Dirjen Imigrasi",
]


async def extract_and_ingest(dry_run: bool = False) -> None:
    """Extract immigration chunks from legal_unified and ingest to immigration_circulars."""
    qd = AsyncQdrantClient(
        url=os.environ["QDRANT_URL"],
        api_key=os.environ["QDRANT_API_KEY"],
    )

    source_collection = "legal_unified_hybrid_hybrid"
    target_collection = "immigration_circulars"

    # Get target collection vector config
    target_info = await qd.get_collection(target_collection)
    logger.info(f"Target {target_collection}: {target_info.points_count} points before enrichment")

    # Scroll source collection looking for immigration-relevant chunks
    matched_points: list[PointStruct] = []
    seen_texts: set[str] = set()
    offset = None
    scanned = 0

    while True:
        points, offset = await qd.scroll(
            collection_name=source_collection,
            limit=200,
            offset=offset,
            with_payload=True,
            with_vectors=True,
        )

        if not points:
            break

        scanned += len(points)

        for point in points:
            text = point.payload.get("text", point.payload.get("content", ""))
            text_lower = text.lower()

            # Check if this chunk is immigration-relevant
            keyword_matches = [kw for kw in IMMIGRATION_KEYWORDS if kw.lower() in text_lower]

            if len(keyword_matches) >= 2:  # At least 2 keyword matches for relevance
                # Deduplicate by first 100 chars
                dedup_key = text[:100].lower().strip()
                if dedup_key in seen_texts:
                    continue
                seen_texts.add(dedup_key)

                # Get dense vector from source
                vectors = point.vector
                if isinstance(vectors, dict) and "dense" in vectors:
                    dense_vector = vectors["dense"]
                elif isinstance(vectors, list):
                    dense_vector = vectors
                else:
                    continue

                metadata = point.payload.get("metadata", {})
                new_point = PointStruct(
                    id=str(uuid.uuid4()),
                    vector={"dense": dense_vector},
                    payload={
                        "text": text,
                        "metadata": {
                            "source_collection": source_collection,
                            "source_point_id": str(point.id),
                            "book_title": metadata.get("book_title", ""),
                            "legal_type": metadata.get("legal_type", ""),
                            "legal_number": metadata.get("legal_number", ""),
                            "legal_year": metadata.get("legal_year", ""),
                            "legal_topic": metadata.get("legal_topic", ""),
                            "keyword_matches": ", ".join(keyword_matches),
                            "doc_type": "immigration_circular",
                        },
                    },
                )
                matched_points.append(new_point)

        if offset is None:
            break

        if scanned % 1000 == 0:
            logger.info(f"  Scanned {scanned} points, matched {len(matched_points)} so far...")

    logger.info(f"Scanned {scanned} total, matched {len(matched_points)} immigration chunks")

    if dry_run:
        logger.info(f"[DRY RUN] Would upsert {len(matched_points)} points to {target_collection}")
        for p in matched_points[:5]:
            meta = p.payload.get("metadata", {})
            logger.info(f"  {meta.get('book_title', '?')[:60]} — keywords: {meta.get('keyword_matches', '')[:40]}")
        return

    # Batch upsert
    batch_size = 100
    for i in range(0, len(matched_points), batch_size):
        batch = matched_points[i:i + batch_size]
        await qd.upsert(collection_name=target_collection, points=batch)
        logger.info(f"  Upserted batch {i//batch_size + 1} ({len(batch)} points)")

    final_info = await qd.get_collection(target_collection)
    logger.info(f"Done. {target_collection}: {target_info.points_count} → {final_info.points_count} points")


if __name__ == "__main__":
    dry_run = "--dry-run" in sys.argv
    asyncio.run(extract_and_ingest(dry_run=dry_run))
```

```python
"""
NLM Async Verification Service

Fire-and-forget background task that verifies Zantara's response against NLM.
NOT in the hot path — runs after response is sent to the user.

Trigger: critical domain queries (visa, tax, legal) with evidence score 0.15-0.60.
Rate limit: max 10 verifications per hour.
Timeout: 30s per NLM call.
"""

from __future__ import annotations

import asyncio
import logging
import os
import time
from typing import Any

import asyncpg

logger = logging.getLogger(__name__)

# Rate limiting
_verification_timestamps: list[float] = []
MAX_VERIFICATIONS_PER_HOUR = 10
NLM_TIMEOUT = 30

# Critical domains that trigger verification
CRITICAL_DOMAINS = {"visa", "tax", "legal", "immigration"}

# NLM notebook mapping by domain
DOMAIN_NOTEBOOK_MAP: dict[str, str] = {
    "visa": "cff93ab0-813a-42f2-a8de-36987e724271",  # NB-2
    "immigration": "cff93ab0-813a-42f2-a8de-36987e724271",  # NB-2
    "tax": os.environ.get("NB_TAX_ID", ""),  # NB-4
    "legal": os.environ.get("NB_LEGAL_ID", ""),  # NB-3
}


def should_verify(domain: str, evidence_score: float, is_greeting: bool, is_pricing: bool) -> bool:
    """Determine if a response should be verified against NLM."""
    if is_greeting or is_pricing:
        return False
    if domain not in CRITICAL_DOMAINS:
        return False
    if evidence_score < 0.15 or evidence_score > 0.60:
        return False

    # Rate limit check
    now = time.time()
    cutoff = now - 3600  # 1 hour window
    recent = [t for t in _verification_timestamps if t > cutoff]
    if len(recent) >= MAX_VERIFICATIONS_PER_HOUR:
        logger.debug("NLM verification rate limit reached, skipping")
        return False

    return True


async def verify_response(
    query: str,
    zantara_answer: str,
    domain: str,
    evidence_score: float,
    db_pool: asyncpg.Pool | None = None,
) -> None:
    """
    Background verification of Zantara response against NLM.
    Logs discrepancies for human review.
    """
    from apps.evaluator.nlm_deep_research.nlm_bridge import nlm_query

    nb_id = DOMAIN_NOTEBOOK_MAP.get(domain, "")
    if not nb_id:
        logger.debug(f"No NLM notebook configured for domain: {domain}")
        return

    _verification_timestamps.append(time.time())

    try:
        result = nlm_query(
            notebook_id=nb_id,
            query=query,
            timeout=NLM_TIMEOUT,
        )

        if result.get("status") != "success" or not result.get("answer"):
            logger.debug(f"NLM verification: no answer for '{query[:40]}...'")
            return

        nlm_answer = result["answer"]

        # Simple discrepancy detection: check if key facts differ
        # Future: use LLM-based comparison
        discrepancy = _detect_discrepancy(zantara_answer, nlm_answer)

        if discrepancy and db_pool:
            await _log_discrepancy(
                db_pool, query, zantara_answer, nlm_answer,
                domain, evidence_score, discrepancy,
            )
            logger.warning(f"NLM verification DISCREPANCY: {query[:40]}... — {discrepancy}")
        else:
            logger.info(f"NLM verification OK: {query[:40]}...")

    except asyncio.TimeoutError:
        logger.debug(f"NLM verification timeout: {query[:40]}...")
    except Exception as e:
        logger.debug(f"NLM verification error: {e}")


def _detect_discrepancy(zantara: str, nlm: str) -> str | None:
    """Basic discrepancy detection between two answers."""
    z_lower = zantara.lower()
    n_lower = nlm.lower()

    # Check for contradictory price/percentage mentions
    import re
    z_numbers = set(re.findall(r'\d+(?:\.\d+)?%|\d{1,3}(?:[.,]\d{3})*', z_lower))
    n_numbers = set(re.findall(r'\d+(?:\.\d+)?%|\d{1,3}(?:[.,]\d{3})*', n_lower))

    if z_numbers and n_numbers:
        diff = z_numbers.symmetric_difference(n_numbers)
        if len(diff) > len(z_numbers.union(n_numbers)) * 0.5:
            return f"Number mismatch: zantara={z_numbers}, nlm={n_numbers}"

    # Check answer length disparity (NLM much longer = missing info in Zantara)
    if len(nlm) > len(zantara) * 3 and len(zantara) < 200:
        return "Zantara answer much shorter than NLM grounded answer"

    return None


async def _log_discrepancy(
    pool: asyncpg.Pool,
    query: str,
    zantara_answer: str,
    nlm_answer: str,
    domain: str,
    evidence_score: float,
    discrepancy: str,
) -> None:
    """Log discrepancy to nlm_verification_log table."""
    try:
        async with pool.acquire() as conn:
            await conn.execute(
                """
                INSERT INTO nlm_verification_log
                    (query, zantara_answer, nlm_answer, domain, evidence_score, discrepancy, created_at)
                VALUES ($1, $2, $3, $4, $5, $6, NOW())
                """,
                query, zantara_answer[:2000], nlm_answer[:2000],
                domain, evidence_score, discrepancy,
            )
    except Exception as e:
        # Table may not exist yet — that's OK, just log
        logger.debug(f"Could not log discrepancy (table may not exist): {e}")
```
