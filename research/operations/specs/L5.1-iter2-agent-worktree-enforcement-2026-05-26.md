### 2.1 PreToolUse hook (Bash) — `worktree_isolation.py`

**Path**: `~/.claude/hooks/worktree_isolation.py`
**Matcher**: `Bash`
**Registered in**: `~/.claude/settings.json` under `hooks.PreToolUse[matcher="Bash"]`

**Logic (iter-2, post-A1+A3+A4+A6)**:

```python
#!/usr/bin/env python3
"""PreToolUse(Bash) hook — block git ops outside allowed worktree.

ITER-2 changes vs ITER-1:
- A1: replace regex with pathlib path canonicalization
- A3: parse `git -C <path>` and `cd <path> && git` effective target
- A4: remove transcript marker bypass
- A6: cache n_alive_agents (zero subprocess hot-path)
"""
import json, os, sys, re, pathlib, subprocess

REPO_ROOT = "/Users/nuzantara/Desktop/nuzantara"
ALIVE_CACHE = pathlib.Path("/tmp/nuz_alive_count")
BLOCKS_FILE = pathlib.Path("/tmp/nuz_l5_1_blocks")

# Patterns for git ops that need worktree isolation (path-canonicalized)
RISKY_GIT_OPS = re.compile(
    r"\bgit\s+(?:-c\s+\S+\s+)*"
    r"(?:-C\s+\S+\s+)?"  # allow -C prefix, capture separately below
    r"(?:checkout|stash|reset|merge|rebase|pull)\b"
)

GIT_C_PATTERN = re.compile(r"\bgit\s+(?:-c\s+\S+\s+)*-C\s+(\S+)")
GIT_CD_PATTERN = re.compile(r"\bcd\s+(\S+)\s*(?:&&|;)\s*git\b")


def env_kill_switch_active() -> bool:
    return os.environ.get("AGENT_WORKTREE_ENFORCEMENT", "true").lower() == "false"


def _git_worktree_list() -> list[dict]:
    """Parse `git worktree list --porcelain` for allowed worktrees."""
    try:
        out = subprocess.run(
            ["git", "-C", REPO_ROOT, "worktree", "list", "--porcelain"],
            capture_output=True, text=True, timeout=2
        ).stdout
    except Exception:
        return []
    worktrees, cur = [], {}
    for line in out.splitlines():
        if line.startswith("worktree "):
            if cur: worktrees.append(cur)
            cur = {"path": line[9:]}
        elif line.startswith("HEAD "):
            cur["head"] = line[5:]
    if cur: worktrees.append(cur)
    return worktrees


def _allowed_worktree_roots() -> list[pathlib.Path]:
    """All paths that are allowed worktree contexts (NOT REPO_ROOT itself)."""
    repo_real = pathlib.Path(REPO_ROOT).resolve()
    roots = [
        repo_real / ".worktrees",   # internal worktree convention (L1)
        repo_real / ".claude" / "worktrees",  # agent subdir convention
    ]
    # Add external worktrees discovered via git worktree list
    for w in _git_worktree_list():
        wp = pathlib.Path(w["path"]).resolve()
        if wp != repo_real:  # exclude main
            roots.append(wp)
    return roots


def _is_path_in_allowed_worktree(target: str) -> bool:
    """A1: path canonicalization replaces broken regex."""
    try:
        target_real = pathlib.Path(target).resolve()
    except Exception:
        return False  # malformed = deny
    repo_real = pathlib.Path(REPO_ROOT).resolve()
    if target_real == repo_real:
        return False  # exactly main = blocked
    if not target_real.is_relative_to(repo_real.parent):
        return True  # outside REPO tree entirely (e.g. /tmp) = allow
    for r in _allowed_worktree_roots():
        if target_real == r or (r.exists() and target_real.is_relative_to(r)):
            return True
    return False


def _effective_git_target(cmd: str, cwd: str) -> str:
    """A3: parse `git -C <path>` or `cd <path> && git`, fallback to cwd."""
    m = GIT_C_PATTERN.search(cmd)
    if m: return m.group(1)
    m = GIT_CD_PATTERN.search(cmd)
    if m: return m.group(1)
    return cwd


def n_alive_cached() -> int:
    """A6: read cache, no subprocess."""
    try:
        return int(ALIVE_CACHE.read_text().strip())
    except Exception:
        return -1


def _increment_blocks_counter():
    """For A9 harmonization with T2.6."""
    try:
        cur = int(BLOCKS_FILE.read_text().strip()) if BLOCKS_FILE.exists() else 0
        BLOCKS_FILE.write_text(str(cur + 1))
    except Exception:
        pass


def main():
    if env_kill_switch_active():
        sys.exit(0)

    payload = json.load(sys.stdin)
    cmd = payload.get("tool_input", {}).get("command", "")
    cwd = payload.get("tool_input", {}).get("cwd") or payload.get("cwd", "")

    if not RISKY_GIT_OPS.search(cmd):
        sys.exit(0)  # not a risky git op

    target = _effective_git_target(cmd, cwd)
    if _is_path_in_allowed_worktree(target):
        sys.exit(0)  # explicit other-worktree work or outside repo

    # Block
    _increment_blocks_counter()
    sys.stderr.write(
        f"WORKTREE ISOLATION VIOLATION (git op)\n"
        f"  command: {cmd[:200]}\n"
        f"  effective target: {target}\n"
        f"  resolved: {pathlib.Path(target).resolve()}\n"
        f"  n_alive_agents: {n_alive_cached()}\n"
        f"  reason: git op in main checkout while sibling agents active\n\n"
        f"Resolution options:\n"
        f"  1. cd into a worktree (`python scripts/agent_start.py --lane X --task-id Y`)\n"
        f"  2. Use `git -C <worktree-path>` explicitly\n"
        f"  3. Set `AGENT_WORKTREE_ENFORCEMENT=false` env var (NOT inline)\n"
    )
    sys.exit(2)


if __name__ == "__main__":
    main()
```

### 2.2 NEW PreToolUse hook (Edit|Write|MultiEdit) — `worktree_file_write_check.py`

**Path**: `~/.claude/hooks/worktree_file_write_check.py`

```python
#!/usr/bin/env python3
"""PreToolUse(Edit|Write|MultiEdit) hook — block file writes outside allowed worktree.

ITER-2 NEW (A2): chiude bypass empirico citato in iter-1 §1 (`kg_bridge.py` orphan
created by sibling via Write tool while this session worked in dedicated worktree).
"""
import json, os, sys, pathlib

REPO_ROOT = "/Users/nuzantara/Desktop/nuzantara"
ALIVE_CACHE = pathlib.Path("/tmp/nuz_alive_count")
BLOCKS_FILE = pathlib.Path("/tmp/nuz_l5_1_blocks")


def env_kill_switch_active() -> bool:
    return os.environ.get("AGENT_WORKTREE_ENFORCEMENT", "true").lower() == "false"


# Re-use logic from worktree_isolation.py (shared module candidate for refactor)
# For minimal-deps hook isolation, copy the 3 functions here.
def _git_worktree_list() -> list[dict]:
    """[shared with worktree_isolation.py — extract to common module if drift]"""
    import subprocess
    try:
        out = subprocess.run(
            ["git", "-C", REPO_ROOT, "worktree", "list", "--porcelain"],
            capture_output=True, text=True, timeout=2
        ).stdout
    except Exception:
        return []
    worktrees, cur = [], {}
    for line in out.splitlines():
        if line.startswith("worktree "):
            if cur: worktrees.append(cur)
            cur = {"path": line[9:]}
    if cur: worktrees.append(cur)
    return worktrees


def _allowed_file_path(file_path: str) -> bool:
    """File write allowed iff target is under an allowed worktree OR outside REPO_ROOT entirely."""
    try:
        fp_real = pathlib.Path(file_path).resolve()
    except Exception:
        return False
    repo_real = pathlib.Path(REPO_ROOT).resolve()

    # Outside REPO_ROOT entirely (e.g. /tmp, ~/.claude, /etc) = allow
    if not fp_real.is_relative_to(repo_real):
        return True

    # Inside REPO_ROOT — must be under allowed worktree
    allowed_roots = [
        repo_real / ".worktrees",
        repo_real / ".claude" / "worktrees",
    ]
    for w in _git_worktree_list():
        wp = pathlib.Path(w["path"]).resolve()
        if wp != repo_real:
            allowed_roots.append(wp)

    for r in allowed_roots:
        if r.exists() and fp_real.is_relative_to(r):
            return True
    return False


def n_alive_cached() -> int:
    try:
        return int(ALIVE_CACHE.read_text().strip())
    except Exception:
        return -1


def _increment_blocks_counter():
    try:
        cur = int(BLOCKS_FILE.read_text().strip()) if BLOCKS_FILE.exists() else 0
        BLOCKS_FILE.write_text(str(cur + 1))
    except Exception:
        pass


def main():
    if env_kill_switch_active():
        sys.exit(0)

    payload = json.load(sys.stdin)
    tool_input = payload.get("tool_input", {})
    file_path = tool_input.get("file_path", "")

    if not file_path:
        sys.exit(0)  # MultiEdit aggregate may have no top-level file_path

    if _allowed_file_path(file_path):
        sys.exit(0)

    # Block
    _increment_blocks_counter()
    sys.stderr.write(
        f"WORKTREE ISOLATION VIOLATION (file write)\n"
        f"  file_path: {file_path}\n"
        f"  resolved: {pathlib.Path(file_path).resolve()}\n"
        f"  n_alive_agents: {n_alive_cached()}\n"
        f"  reason: writing to main checkout while sibling agents active\n\n"
        f"Resolution options:\n"
        f"  1. cd into a worktree first (`python scripts/agent_start.py --lane X --task-id Y`)\n"
        f"  2. Set `AGENT_WORKTREE_ENFORCEMENT=false` env var\n"
        f"  3. If this is operator-direct-edit on main (cicatrix hotfix etc.), use kill switch\n"
    )
    sys.exit(2)


if __name__ == "__main__":
    main()
```
