### Step 4 — Write 6 consolidated scripts

```bash
# Block 1 — Machine + Environment

ssh -o ConnectTimeout=2 $OTHER 'echo "Peer: $(whoami)@$(hostname)"' 2>/dev/null || echo "Peer: UNREACHABLE"

# Python

  echo "Backend venv: $(~/Desktop/nuzantara/apps/backend-rag/.venv/bin/python --version)"
fi

# Brew services
brew services list 2>/dev/null | grep -E "postgresql|redis|qdrant" | awk '{print $1": "$2}'

exit 0
```

```bash
# Capture OLD output (before refactor) — DO THIS FIRST during Step 0 audit
# OLD: ls ~/backups/old-sessionstart-output.txt

# After refactor, capture NEW output

# 13 critical keywords MUST be present in NEW output
CRITICAL_KEYWORDS=(
    "Autonomous Ops"          # Block 2
    "Symbiosis\|SYMBIOSIS"    # Block 2
    "Anti-halluc\|hallucinat" # Block 3 lessons
    "4-LLM\|panel"            # Block 3 lessons

    "Peer:"                   # Block 1

    "Resume\|BALI ZERO"       # Block 6

)

MISSING=0
for kw in "${CRITICAL_KEYWORDS[@]}"; do
    if echo "$NEW_OUTPUT" | grep -qE "$kw"; then
        echo "✅ $kw"
    else
        echo "❌ MISSING: $kw"
        ((MISSING++))
    fi
done

if [ $MISSING -gt 0 ]; then
    echo "ABORT: $MISSING critical keyword(s) missing — rollback required"
    exit 1
else
    echo "PASS: all 13 keywords present in new SessionStart"
fi

# Additionally: SHA256 hash check on info-preserving substrings
# (catches subtle compression that drops detail)
for kw in "${CRITICAL_KEYWORDS[@]}"; do
    OLD_HASH=$(grep -E "$kw" ~/backups/old-sessionstart-output.txt 2>/dev/null | shasum -a 256 | head -c 16)
    NEW_HASH=$(echo "$NEW_OUTPUT" | grep -E "$kw" | shasum -a 256 | head -c 16)
    echo "$kw: old=$OLD_HASH new=$NEW_HASH"
done
```
