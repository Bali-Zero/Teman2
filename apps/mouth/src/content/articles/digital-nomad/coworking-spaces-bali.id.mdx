  title="Anggaran Coworking"

      ],

    },
    {
      id: "tier",
      label: "Kualitas Ruang",
      type: "select",
      options: [
        { value: "budget", label: "Ramah budget" },
        { value: "standard", label: "Coworking standar" },
        { value: "premium", label: "Premium/kantor pribadi" }
      ],
      defaultValue: "standard"
    }
  ]}
  calculateResult={(values) => {
    // Pricing estimates
    const pricing = {
      budget: { monthly: 1500000, day: 100000 },
      standard: { monthly: 2500000, day: 150000 },
      premium: { monthly: 4000000, day: 250000 }
    };

    const tier = pricing[values.tier];
    let monthly;

    switch (values.frequency) {
      case "daily":
        monthly = tier.monthly;
        break;
      case "parttime":
        monthly = Math.min(tier.day * 16, tier.monthly);
        break;
      case "occasional":
        monthly = tier.day * 8;
        break;
      case "dropins":
        monthly = tier.day * 4;
        break;
    }

    return {
      total: monthly,
      breakdown: [
        { label: "Biaya Ruang Kerja", amount: monthly },
        { label: "Kopi/Minuman", amount: 300000 },
        { label: "Makan Siang Sesekali", amount: 500000 }
      ],
      currency: "IDR",
      note: `Sekitar USD ${Math.round(monthly / 16000)}/bulan`
    };

}}

**Pro:**

- Tidak ada perjalanan harian

## Tips untuk Digital Nomad

1. **Coba sebelum berkomitmen** - Beli day pass dulu
2. **Cek internet cadangan** - Tanyakan apa yang terjadi ketika utama gagal
3. **Kunjungi selama jam kerja Anda** - Cek apakah terlalu ramai
4. **Tes kualitas panggilan video** - Penting untuk sebagian besar kerja remote
5. **Pertimbangkan lokasi** - Dekat akomodasi menghemat waktu
6. **Manfaatkan komunitas** - Networking berharga
7. **Miliki cadangan mobile** - SIM Telkomsel untuk darurat
8. **Pesan ruang rapat awal** - Ruang populer penuh cepat
