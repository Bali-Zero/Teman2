<Calculator
  title="Анализ инвестиций в виллы"
  description="Оценка инвестиционных затрат и доходности"
  fields={[
    {

      type: "number",
      defaultValue: 5000000000,

        { value: "4", label: "4 спальни" },
        { value: "5", label: "5+ спален" }
      ],
      defaultValue: "3"
    },
    {
      id: "location",
      label: "Местоположение",
      type: "select",
      options: [
        { value: "premium", label: "Премиум (Семиньяк, центр Чангу)" },
        { value: "good", label: "Хорошее (Переренан, Умалас)" },
        { value: "developing", label: "Развивающееся (Табанан, Север)" }
      ],
      defaultValue: "good"
    },
    {
      id: "occupancy",
      label: "Ожидаемая загрузка %",
      type: "number",
      defaultValue: 60,
      min: 20,
      max: 90
    }
  ]}
  calculateResult={(values) => {
    const purchase = Number(values.purchasePrice);
    const bedrooms = Number(values.bedrooms);
    const occupancy = Number(values.occupancy) / 100;

    // Nightly rates by location and bedrooms
    const rateMap = {
      "premium-2": 3000000, "premium-3": 4500000, "premium-4": 6000000, "premium-5": 8000000,
      "good-2": 2000000, "good-3": 3000000, "good-4": 4000000, "good-5": 5500000,
      "developing-2": 1200000, "developing-3": 1800000, "developing-4": 2500000, "developing-5": 3500000
    };

    const nightlyRate = rateMap[`${values.location}-${values.bedrooms}`] || 2500000;
    const annualGross = nightlyRate * 365 * occupancy;

    // Operating costs
    const management = annualGross * 0.20; // 20% management
    const maintenance = 60000000; // General maintenance
    const utilities = 36000000; // Utilities avg
    const marketing = 24000000; // OTA fees, marketing
    const insurance = 12000000;
    const tax = annualGross * 0.10; // Rental income tax

    const totalCosts = management + maintenance + utilities + marketing + insurance + tax;
    const netIncome = annualGross - totalCosts;
    const grossYield = (annualGross / purchase) * 100;
    const netYield = (netIncome / purchase) * 100;

    return {
      total: netIncome,
      breakdown: [
        { label: "Валовой годовой доход", amount: annualGross },
        { label: "Управление (20%)", amount: -management },
        { label: "Обслуживание", amount: -maintenance },
        { label: "Коммунальные услуги", amount: -utilities },
        { label: "Маркетинг/OTA", amount: -marketing },
        { label: "Страхование", amount: -insurance },
        { label: "Налог на аренду (10%)", amount: -tax },
        { label: "Чистый годовой доход", amount: netIncome }
      ],
      currency: "IDR",
      note: `Валовая доходность: ${grossYield.toFixed(1)}% | Чистая доходность: ${netYield.toFixed(1)}%`
    };

}}

## Лицензирование аренды

<InfoCard variant="warning" title="Требуется лицензия на аренду">
Для эксплуатации арендной виллы требуется:
- Лицензия **Pondok Wisata** для небольших гостиниц
- **TDUP** для более крупных операций
- Туристическая регистрация
- Различные местные разрешения

Работа без лицензий грозит штрафами, закрытием и проблемами с визами.

</InfoCard>

### Требования к лицензии

| Лицензия      | Для                     | Выдается               |
| ------------- | ----------------------- | ---------------------- |
| Pondok Wisata | Небольшие гостевые дома | Местное самоуправление |
| TDUP          | Туристическое жилье     | Отдел туризма          |
| NIB           | Регистрация бизнеса     | OSS                    |
| Экологическая | Воздействие операций    | Отдел окружающей среды |
